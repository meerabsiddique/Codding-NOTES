Dynamic 2D Array — Full Template
// ALLOCATE
int** allocate(int& r, int& c) {
    cin >> r >> c;
    int** m = new int*[r];
    for(int i=0; i<r; i++)
        m[i] = new int[c]();  // () = zero init
    return m;
}

// PRINT
void display(int** m, int r, int c) {
    for(int i=0;i<r;i++){
        for(int j=0;j<c;j++)
            cout<<m[i][j]<<" ";
        cout<<endl;
    }
}

// DEALLOCATE
void deallocate(int** m, int r) {
    for(int i=0;i<r;i++)
        delete[] m[i];   // inner first
    delete[] m;           // then outer
    m = nullptr;
}
Abstract Class + Polymorphism Template
class Animal {
protected:
    char* name;
public:
    Animal(const char* n){
        name = new char[strlen(n)+1];
        strcpy(name, n);
    }
    virtual void display() = 0; // pure virtual
    virtual ~Animal() { delete[] name; }
};

class Bird : public Animal {
    string color;
public:
    Bird(const char* n, string c)
        : Animal(n), color(c) {}
    void display() override {
        cout << name << " chirps\n";
    }
};

// In main — array of base pointers
Animal* arr[3];
arr[0] = new Bird("Parrot", "green");
arr[0]->display(); // polymorphic call
delete arr[0];    // calls correct destructor
Operator Overloading — Key Patterns
class MyClass {
    int* data; int size;
public:
    // + operator (returns new object)
    MyClass operator+(MyClass& other);

    // [] subscript
    int& operator[](int i){ return data[i]; }

    // prefix ++
    MyClass& operator++();

    // postfix ++ (dummy int)
    MyClass operator++(int);

    // == comparison
    bool operator==(MyClass& o);

    // unary minus (pop / negate)
    MyClass operator-();

    // << and >> MUST be friend
    friend ostream& operator<<
        (ostream& os, const MyClass& obj);
    friend istream& operator>>
        (istream& is, MyClass& obj);
};
Postfix++: save copy BEFORE increment, return the copy
char* Member — Always Do This
// Constructor — deep copy
Item(const char* n) {
    name = new char[strlen(n)+1];
    strcpy(name, n);
}

// Copy constructor
Item(const Item& o) {
    name = new char[strlen(o.name)+1];
    strcpy(name, o.name);
}

// Assignment operator
Item& operator=(const Item& o) {
    if(this != &o) {       // self-check!
        delete[] name;
        name = new char[strlen(o.name)+1];
        strcpy(name, o.name);
    }
    return *this;
}

// Destructor
~Item() { delete[] name; name=nullptr; }
Never use = to copy char* — always strcpy
Template Function + Class
// Template function
template <typename T>
T findMax(T** arr, int r, int c) {
    T max = arr[0][0];
    for(int i=0;i<r;i++)
        for(int j=0;j<c;j++)
            if(arr[i][j] > max) max=arr[i][j];
    return max;
}

// Template class
template <class T>
class Stack {
    T* data; int top, size;
public:
    Stack(int s): size(s), top(-1)
        { data = new T[s]; }
    void push(T val){ data[++top]=val; }
    T    pop()       { return data[top--]; }
    bool isEmpty()  { return top==-1; }
    ~Stack(){ delete[] data; }
};

Composition vs Aggregation — Quick
// COMPOSITION — Shelf owns Media objects
class Shelf {
    Media** items; // owns them
    int currSize, maxSize;
public:
    Shelf(int max): maxSize(max), currSize(0){
        items = new Media*[maxSize];
    }
    void insert(Media* m){ items[currSize++]=m; }
    ~Shelf(){
        for(int i=0;i<currSize;i++)
            delete items[i]; // delete each item
        delete[] items;
    }
};

// AGGREGATION — Enclosure uses Zookeeper
class Enclosure {
    Zookeeper* zk; // does NOT own it
public:
    Enclosure(Zookeeper* z): zk(z) {}
    ~Enclosure(){  // do NOT delete zk
    }
};

🎯 Critical Tricks & Common Mistakes
Memory — Never Forget
  • Delete inner before outer in 2D arrays
  • Always set pointer to nullptr after delete
  • Deep copy in copy constructor — never shallow
  • Check this != &other in operator=
  • Virtual destructor in every base class with pointers
  • delete for single object, delete[] for arrays
Polymorphism Rules
  • Must have virtual for runtime dispatch
  • Pure virtual = = 0 → class becomes abstract
  • Override in child: same signature + override keyword
  • dynamic_cast needs virtual functions to work
  • Base pointer can hold derived object
  • dynamic_cast → nullptr on pointer fail, bad_cast on ref fail
Quick Fixes When Stuck
  • Compile error on <</>> → add friend
  • Can't call derived function via base ptr → check virtual
  • Crash on delete → probably double-delete or dangling ptr
  • Wrong copy → use strcpy not =
  • Template not found → define in .h not .cpp
  • Abstract class object error → check all pure virtuals are overridden
Initializer List — Use Always
// Instead of this (assignments in body)
Car(string b, int hp) {
    brand = b;       // assignment
    engine = Engine(hp); // creates then copies
}

// Do THIS (initializer list — constructs directly)
Car(string b, int hp)
    : brand(b), engine(hp)  // engine constructed once
{
    cout << "Car ready\n";
}

// MUST use initializer list for:
// - const members
// - reference members
// - objects without default constructor
vector<T*> vs T** — When to Use
// vector (modern, 2023 paper used this)
#include <vector>
vector<Animal*> animals;
animals.push_back(new Bird(...));
animals.size();       // count
for(auto a : animals) a->display();

// Manual T** (older papers, more common)
Animal** items = new Animal*[maxSize];
int count = 0;
items[count++] = new Bird(...);
// delete loop needed in destructor
for(int i=0;i<count;i++) delete items[i];
delete[] items;
Both appear in papers — know both patterns
